Data type	Example	                                        Meaning
int	        age = 30	                                    Integer
float	    temp = 36.5	                                    Floating-point number
complex	    x = 2 + 3j	                                    Complex number
str	        name = "Thushantha"	                            Text/string
bool	    connected = True	                            Boolean: True or False
list	    ports = [22, 80, 443]	                        Ordered, changeable collection
tuple	    point = (10, 20)	                            Ordered, immutable collection
dict	    server = {"ip": "192.168.1.220","port": 22}	    Key/value pairs
set	        ports = {22, 80, 443}	                        Unique unordered values
frozenset	x = frozenset({1, 2, 3})	                    Immutable set
range	    range(1, 10)	                                Sequence of integers
bytes	    data = b"Hello"	                                Immutable binary data
bytearray	data = bytearray(10)	                        Mutable binary data
memoryview	memoryview(data)	                            View into binary data
NoneType	result = None	                                Represents no value
